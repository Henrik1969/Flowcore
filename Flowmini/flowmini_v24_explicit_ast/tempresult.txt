program bubblesort_compound

main {
    arr  : list<int>([1,9,3,5,8,2,2,6,4])
    len  : int(length(arr))
    zero : int(0)
    one  : int(1)
    i    : int(0)
    j    : int(0)
    tmp  : int(0)

    while i < len {
        zero -> j

        while j + one < len {
            if arr[j] > arr[j + one] {
                arr[j] -> tmp
                arr[j + one] -> arr[j]
                tmp -> arr[j + one]
            }

            j + one -> j
        }

        i + one -> i
    }

    print arr
}

program bubblesort_indexed_structured

main {
    arr   : list<int>([1,9,3,5,8,2,2,6,4])
    len   : int(length(arr))
    zero  : int(0)
    one   : int(1)
    i     : int(0)
    j     : int(0)
    next  : int(0)
    a     : int(0)
    b     : int(0)

    while i < len {
        zero -> j

        while j < len {
            j + one -> next

            if next < len {
                arr[j] -> a
                arr[next] -> b

                if a > b {
                    b -> arr[j]
                    a -> arr[next]
                }
            }

            j + one -> j
        }

        i + one -> i
    }

    print arr
}

program capable

fn identity(x : int): int {
    x -> return
}

fn add(a : int, b : int): int {
    a + b -> return
}

fn square(n : int): int {
    n * n -> return
}

fn add_then_square(a : int, b : int): int {
    sum : int(0)
    add(a, b) -> sum
    square(sum) -> return
}

fn factorial(n : int): int {
    one : int(1)
    acc : int(1)

    while n > one {
        acc * n -> acc
        n - one -> n
    }

    acc -> return
}

fn collatz_next(n : int): int {
    zero  : int(0)
    two   : int(2)
    three : int(3)
    one   : int(1)
    rem   : int(0)

    n % two -> rem

    if rem == zero {
        n / two -> return
    } else {
        n * three + one -> return
    }
}

fn choose_larger(a : int, b : int): int {
    if a > b {
        a -> return
    } else {
        b -> return
    }
}

main {
    zero  : int(0)
    one   : int(1)
    two   : int(2)
    three : int(3)
    four  : int(4)
    five  : int(5)
    eight : int(8)
    nine  : int(9)
    ten   : int(10)
    forty_two : int(42)

    result : int(0)

    identity(forty_two) -> result
    print result

    add_then_square(three, four) -> result
    print result

    square(add(three, four) + one) -> result
    print result

    print result

    collatz_next(five) -> result
    print result

    choose_larger(nine, four) -> result
    print result

    arr      : list<int>([1,9,3,5,8,2])
    len      : int(length(arr))
    i        : int(0)
    value    : int(0)
    rem      : int(0)
    list_sum : int(0)
    odd_sum  : int(0)
    first_even : int(0)

    while i < len {
        arr[i] -> value
        list_sum + value -> list_sum
        i + one -> i
    }

    print list_sum

    zero -> i

    while i < len {
        arr[i] -> value
        value % two -> rem

        if rem == zero {
            i + one -> i
            continue
        }

        odd_sum + value -> odd_sum
        i + one -> i
    }

    print odd_sum

    zero -> i

    while i < len {
        arr[i] -> value
        value % two -> rem

        if rem == zero {
            value -> first_even
            break
        }

        i + one -> i
    }

    print first_even

    arr[one] -> value
    print value

    forty_two -> arr[one]
    arr[one] -> value
    print value
    print arr

    matrix : array<int>[2, 3]([1,2,3,4,5,6])
    row    : int(1)
    col    : int(2)

    matrix[row, col] -> value
    print value

    value + ten -> matrix[row, col]
    print matrix

    cube : array<int>[2, 2, 2]([1,2,3,4,5,6,7,8])
    xidx : int(1)
    yidx : int(0)
    zidx : int(1)

    cube[xidx, yidx, zidx] -> value
    print value

    ma : array<int>[2, 2]([1,2,3,4])
    mb : array<int>[2, 2]([5,6,7,8])
    mc : array<int>[2, 2]([0,0,0,0])
    mi : int(0)
    mj : int(0)
    mk : int(0)
    cell_sum : int(0)

    while mi < two {
        zero -> mj

        while mj < two {
            zero -> mk
            zero -> cell_sum

            while mk < two {
                cell_sum + ma[mi, mk] * mb[mk, mj] -> cell_sum
                mk + one -> mk
            }

            cell_sum -> mc[mi, mj]
            mj + one -> mj
        }

        mi + one -> mi
    }

    print mc
}
